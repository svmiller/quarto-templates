% Allow for those simple two-tone footlines I like.
% Edit the colors as you see fit. I forget where I stole this from.
%------------------------------------------------------------------
\defbeamertemplate*{footline}{my footline}{%
    \ifnum\insertpagenumber=1
    \hbox{%
        \begin{beamercolorbox}[wd=\paperwidth,ht=.8ex,dp=1ex,center]{}%
      % empty environment to raise height
        \end{beamercolorbox}%
    }%
    \vskip0pt%
    \else%
        \Tiny{%
            \hfill%
		\vspace*{1pt}%
            \insertframenumber/\inserttotalframenumber \hspace*{0.1cm}%
            \newline%
            \color{$if(primarycolor)$primarycolor$else$sublue$endif$}{\rule{\paperwidth}{0.4mm}}\newline%
            \color{$if(secondarycolor)$secondarycolor$else$suwater$endif$}{\rule{\paperwidth}{.4mm}}%
        }%
    \fi%
}
